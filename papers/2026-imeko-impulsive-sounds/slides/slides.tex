\section{Motivation}

\begin{frame}{Motivation and research questions}
  \begin{itemize}
    \item Acoustic detection of gunfire complements camera-based surveillance: the
          acoustic channel is omnidirectional and does not require line of sight.
    \item Three required functions, in outdoor environments that are noisy and
          reverberant:
  \end{itemize}
  \vspace{-1mm}
  % the three functions as a quiet icon row
  \begin{center}
  \begin{tikzpicture}
    \fnode{0}{\faWaveSquare}{detection}{of the impulsive event}
    \fnode{4.4cm}{\faTags}{classification}{ideally down to calibre}
    \fnode{8.8cm}{\faMapMarker*}{localization}{of the source, in 2D}
  \end{tikzpicture}
  \end{center}
  \vfill
  \pause
  \begin{block}{This study}
    \begin{enumerate}
      \item Which cepstral representation separates gunshots from acoustically similar
            false alarms most reliably?
      \item Neural network or support vector machine on those features?
      \item How does the analysis frame length interact with environmental reflections?
    \end{enumerate}
  \end{block}
      \note<2>{
\textbf{[0:45--1:50]}~{\color{accent}(1/2)}\par\medskip
    This work is part of a larger project -- an acoustic sensor system for protecting
    soft targets. // Acoustic detection is a good complement to cameras. // Sound is
    omnidirectional, and you don't need a line of sight. //\par\medskip
    {\color{accent}[CUT IF LATE]} The full system has to detect an impulsive event, classify it -- ideally
    down to the calibre -- and localize the source. // And it has to work outdoors, with
    noise and a lot of reflections. {\color{accent}[/CUT]} //
  }
\end{frame}

\note{
\textbf{[0:45--1:50]}~{\color{accent}(2/2)}\par\medskip
    \textbf{[CLICK]}\par\medskip
    In this study we ask three questions. //\par\medskip
    First, which cepstral features best separate gunshots from similar impulsive sounds?
    // Second, which classifier works better -- a neural network or a support vector
    machine? // And third, how does the frame length affect the results? //
  }

\section{Signal}

\begin{frame}{The acoustic signature of a gunshot}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.58\textwidth}
      \vspace{-3mm}
      \includegraphics[width=\textwidth]{figs/gunshot_char.png}
    \end{column}
    \begin{column}{0.40\textwidth}
      \vspace{2mm}
      {\small One or two components, depending on projectile speed:}
      \begin{itemize}\small
        \item \textbf{Shock wave} (``N-wave'') -- \emph{supersonic only} ($v > c$);
              generated along the trajectory, so it arrives first.
        \pause
        \item \textbf{Muzzle blast} -- \emph{always present}; propellant gases at the
              barrel, 1--3\,ms; here $\approx$45\,ms behind.
        \item Between and after: \gtcc{reflections}.
      \end{itemize}
    \end{column}
  \end{columns}
      \note<2>{
\textbf{[1:50--3:00]}~{\color{accent}(1/2)}\par\medskip
    Let's start with the signal of a gunshot. //\par\medskip
    A gunshot has one or two components, depending on the projectile speed. // A
    subsonic shot has only the muzzle blast. // A supersonic one also has a shock wave. //
    This recording is a supersonic shot, so we can see both. //\par\medskip
    The first one, at about two and a half milliseconds, is the shock wave. //
    {\color{accent}[CUT IF LATE]} It's created along the bullet's path and has the typical N-shape.
    {\color{accent}[/CUT]} //
  }
\end{frame}

\note{
\textbf{[1:50--3:00]}~{\color{accent}(2/2)}\par\medskip
    \textbf{[CLICK]}\par\medskip
    The second one, about forty-five milliseconds later, is the muzzle blast. // It comes
    from the gases leaving the barrel, and every gunshot has it. //\par\medskip
    Everything between and after them is mostly reflections from the surroundings. //
    This matters for the frame length, and I'll come back to it in the results. //
  }

\begin{frame}{Single-domain analysis is insufficient}
  \begin{itemize}
    \item \textbf{Spectral analysis alone} often captures environmental noise rather
          than the gunshot itself.
    \item \textbf{Time-domain patterns} get distorted by nonlinear dispersion and
          obstacles.
  \end{itemize}
  \vfill
  \begin{alertblock}{Consequence}
    Effective feature extraction must integrate \emph{both} time and frequency
    domain processing.
  \end{alertblock}
      \note{
\textbf{[3:00--3:30]}\par\medskip
    This also shows why one domain alone is not enough. //\par\medskip
    The spectrum often tells you more about the environment than about the gunshot. //
    And the waveform changes as the sound travels and hits obstacles. //\par\medskip
    So we need features that combine both domains, and cepstral features do exactly
    that. //
  }
\end{frame}

\begin{frame}{Localization -- bearing from TDoA}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.56\textwidth}
      \includegraphics[width=0.96\textwidth]{figs/tdoa.png}
    \end{column}
    \begin{column}{0.42\textwidth}
      \vspace{4mm}
      \[ \Theta = \arcsin\!\left(\frac{c\,\Delta n}{d\,f_s}\right) \]
      \vspace{1mm}
      \begin{itemize}\small
        \item Two MEMS microphones per unit ($d = 0.186$\,m, $f_s = 44.1$\,kHz)
              give one \textbf{bearing} $\Theta$.
        \pause
        \item Two units, two bearings -- they \textbf{intersect} at the event.
        \item Measured: cross-correlation with parabolic interpolation,
              \gtcc{MAE $0.77^\circ$} over 70 events ($-45^\circ$ to $+60^\circ$).
      \end{itemize}
    \end{column}
  \end{columns}
      \note<2>{
\textbf{[4:20--5:20]}~{\color{accent}(1/2)}\par\medskip
    Let me briefly explain the localization first. //\par\medskip
    Each sensor unit has two MEMS microphones, about twenty centimetres apart, sampled
    at 44.1 kilohertz. //\par\medskip
    We measure the time difference of arrival between the two microphones. // From this
    delay we get the direction to the source, using the equation here. //
  }
\end{frame}

\note{
\textbf{[4:20--5:20]}~{\color{accent}(2/2)}\par\medskip
    \textbf{[CLICK]}\par\medskip
    So one unit gives us one direction. // With two units we get two directions, and
    where they cross is the position of the source. //\par\medskip
    We tested this on seventy events at nine reference angles. // With cross-correlation
    and parabolic interpolation, the mean absolute error was zero point seven seven
    degrees. //
  }

\section{Conclusion}

\begin{frame}{Conclusion}
  \begin{itemize}
    \item A distributed network of stand-alone acoustic sensors with a remote server
          detects, classifies and localizes impulsive sounds in 2D.
    \item Cepstral coefficients are an effective feature extractor for gunshot
          classification with both NN and SVM.
    \item GTCC beats Mel-frequency based coefficients; NN achieves slightly better
          classification success, especially with longer frames.
    \item Validated with various firearm calibers and ammunition.

\begin{frame}{Appendix -- other cepstral coefficients}
  \begin{itemize}
    \item \textbf{LPCC} (Linear Prediction Cepstral Coefficients) and \textbf{PLPC}
          (Perceptual Linear Prediction Coefficients) -- noise elimination, speech
          recognition.
    \item \textbf{GFCC} (Greenwood Function Cepstral Coefficients) -- derived from
          human auditory response models, used for classifying animal sounds.
    \item \textbf{GTCC} (Gammatone-Frequency Cepstral Coefficients) -- same family,
          greater resistance to noise than MFCC.
  \end{itemize}
\end{frame}

\begin{frame}{Appendix -- metrics}
  \begin{description}
    \item[ACC] Accuracy -- share of correctly classified events.
    \item[RLC] Recall, also known as \emph{sensitivity} -- share of actual gunshots
      of the class that were detected.
    \item[MCC] Matthews Correlation Coefficient -- balanced measure robust to class
      imbalance; $+1$ perfect, $0$ random.
  \end{description}
  \vfill
  {\small\color{gray} Definitions as used in the source study: Svatoš \& Holub,
   \emph{Meas. Sci. Technol.} \textbf{35} (2024) 076122, eqs.~(1)--(3).}
\end{frame}

\section{Motivation}

\begin{frame}{Motivation and research questions}
  \begin{itemize}
    \item Acoustic detection of gunfire complements camera-based surveillance: the
          acoustic channel is omnidirectional and does not require line of sight.
    \item Required functions: \textbf{detection} of the impulsive event,
          \textbf{classification} (ideally down to calibre), and \textbf{localization}
          of the source -- in outdoor environments that are noisy and reverberant.
  \end{itemize}
  \vfill
  \begin{block}{This study}
    \begin{enumerate}
      \item Which cepstral representation separates gunshots from acoustically similar
            false alarms most reliably?
      \item Neural network or support vector machine on those features?
      \item How does the analysis frame length interact with environmental reflections?
    \end{enumerate}
  \end{block}
  \note{%
    \textbf{[0:45--1:50]}\par\medskip
    This work is part of the development of an acoustic sensor unit for the
    protection of soft targets. //\par\medskip
    Acoustic detection of gunfire complements camera-based surveillance. The acoustic
    channel has two properties cameras lack: it is omnidirectional, and it does not
    require line of sight. //
    {\color{accent}[CUT IF LATE]} The system has to provide three functions --
    detection of the impulsive event, classification, ideally down to calibre, and
    localization of the source -- in outdoor environments that are both noisy and
    reverberant. //\par\medskip
    This study addresses three questions. Which cepstral representation separates
    gunshots from acoustically similar false alarms most reliably. // Whether a neural
    network or a support vector machine performs better on those features. // And how
    the analysis frame length interacts with environmental reflections. //
  }
\end{frame}

\section{Signal}

\begin{frame}{The acoustic signature of a gunshot}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.62\textwidth}
      \vspace{-2mm}
      \includegraphics[width=\textwidth]{figs/gunshot_char.png}
    \end{column}
    \begin{column}{0.36\textwidth}
      \vspace{2mm}
      {\small One or two components, depending on projectile speed:}
      \begin{itemize}\small
        \item \textbf{Shock wave} (``N-wave'') -- \emph{supersonic only} ($v > c$);
              generated along the trajectory, so it arrives first.
        \pause
        \item \textbf{Muzzle blast} -- \emph{always present}; propellant gases at the
              barrel, 1--3\,ms; here $\approx$45\,ms behind.
        \item Between and after: \gtcc{reflections}.
      \end{itemize}
    \end{column}
  \end{columns}
  \note<2>{
    \textbf{[1:50--3:00]}~{\color{accent}(1/2)}\par\medskip
    Both detection and localization depend on what the signal actually looks like, so
    let us start there. //\par\medskip
    A gunshot recording contains one or two components, depending on the speed of the
    projectile. A subsonic shot produces the muzzle blast alone; a supersonic
    projectile adds a shock wave ahead of it. // The recording shown here is a
    supersonic shot, so both are present. //\par\medskip
    The first, at approximately two and a half milliseconds, is the shock wave.
    {\color{accent}[CUT IF LATE]} It is generated along the trajectory as a Mach cone,
    and the portion that reaches the microphone has the characteristic N-shaped
    pressure profile. {\color{accent}[/CUT]} //\par\medskip
  }
\end{frame}

\note{
    \textbf{[1:50--3:00]}~{\color{accent}(2/2)}\par\medskip
    \textbf{[CLICK]}\par\medskip
    The muzzle blast -- in this recording about forty-five milliseconds later -- is the
    expansion of propellant gases at the barrel. This component is present for every
    shot, subsonic or supersonic. It originates at the instant of firing but propagates
    at the speed of sound, which is why it arrives after the shock wave. Its duration is
    one to three milliseconds. //\par\medskip
    The signal between and after the two components is not part of the source
    signature; it consists of reflections from the surroundings. This is relevant to
    the choice of frame length, and I return to it in the results. //
  }

\begin{frame}{Single-domain analysis is insufficient}
  \begin{itemize}
    \item \textbf{Spectral analysis alone} often captures environmental noise rather
          than the gunshot itself.
    \item \textbf{Time-domain patterns} get distorted by nonlinear dispersion and
          obstacles.
  \end{itemize}
  \vfill
  \begin{alertblock}{Consequence}
    Effective feature extraction must integrate \emph{both} time and frequency
    domain processing.
  \end{alertblock}
  \note{
    \textbf{[3:00--3:30]}\par\medskip
    Neither signal domain is sufficient on its own. //\par\medskip
    Spectral analysis of a frame tends to characterise the acoustic environment rather
    than the source. // The time-domain pattern is distorted by nonlinear dispersion
    during propagation and by obstacles in the path. //\par\medskip
    Feature extraction therefore has to combine time- and frequency-domain processing
    -- which is what a cepstral representation does. //
  }
\end{frame}

\begin{frame}{Localization -- bearing from TDoA}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.56\textwidth}
      \includegraphics[width=0.96\textwidth]{figs/tdoa.png}
    \end{column}
    \begin{column}{0.42\textwidth}
      \vspace{4mm}
      \[ \Theta = \arcsin\!\left(\frac{c\,\Delta n}{d\,f_s}\right) \]
      \vspace{1mm}
      \begin{itemize}\small
        \item Two MEMS microphones per unit ($d = 0.186$\,m, $f_s = 44.1$\,kHz)
              give one \textbf{bearing} $\Theta$.
        \pause
        \item Two units, two bearings -- they \textbf{intersect} at the event.
        \item Measured: cross-correlation with parabolic interpolation,
              \gtcc{MAE $0.77^\circ$} over 70 events ($-45^\circ$ to $+60^\circ$).
      \end{itemize}
    \end{column}
  \end{columns}
  \note<2>{
    \textbf{[4:20--5:20]}~{\color{accent}(1/2)}\par\medskip
    Localization. // Each sensor unit carries two MEMS microphones, almost twenty
    centimetres apart, sampled at 44.1 kilohertz. //\par\medskip
    The time difference of arrival between the two channels -- at most about
    twenty-four samples for this geometry -- gives the bearing theta to the source
    through the relation shown: the arcsine of the speed of sound times the delay in
    samples, over the microphone spacing times the sampling frequency. //\par\medskip
  }
\end{frame}

\note{
    \textbf{[4:20--5:20]}~{\color{accent}(2/2)}\par\medskip
    \textbf{[CLICK]}\par\medskip
    A single unit thus yields a direction. Two units yield two bearings, and their
    intersection is the position estimate. //\par\medskip
    Bearing accuracy was evaluated on seventy recorded events at nine reference angles
    between minus forty-five and plus sixty degrees. Cross-correlation with parabolic
    sub-sample interpolation gives a mean absolute error of 0.77 degrees. //
  }

\begin{frame}{Discussion}
  \begin{itemize}
    \item \textbf{MFCC, LFCC and IMFCC yield similar results.}
    \item \gtcc{GTCC consistently performs better} across every frame length and both
          classifiers.
    \item \textbf{NN outperforms SVM} -- better at modelling complex patterns,
          especially with longer frames.
    \item \textbf{Longer frames improve accuracy}, but start to capture environmental
          reflections, which can pull focus away from the gunshot itself.
  \end{itemize}
  \vfill
  \begin{block}{Best configuration in the table}
    GTCC + NN at 50\,ms -- 99.15\,\% ACC, 96.77\,\% RLC, MCC 0.98.
  \end{block}
  \note{
    \textbf{[7:35--8:45]}~{\color{accent}(1/2)}\par\medskip
    Three observations. //\par\medskip
    First, MFCC, LFCC and IMFCC perform comparably. The choice between mel, linear and
    inverted-mel spacing has little effect. //\par\medskip
    Second, GTCC performs consistently better -- at every frame length and for both
    classifiers.
    {\color{accent}[CUT IF LATE]} The gammatone bank is derived from an auditory model
    and is known to be more noise-robust than MFCC; that is the probable explanation,
    but the mechanism was not isolated in this study. {\color{accent}[/CUT]}
    //\par\medskip
  }
\end{frame}

\note{
    \textbf{[7:35--8:45]}~{\color{accent}(2/2)}\par\medskip
    Third, the neural network is slightly ahead of the SVM, particularly at longer
    frames -- a tendency across the study rather than a rule for every row of this
    table. //\par\medskip
    Finally, longer frames improve accuracy, but a fifty-millisecond frame increasingly
    contains the impulse response of the surroundings rather than the source signature
    -- the reflections seen on the recording earlier. The trend cannot be extrapolated
    further. //
  }

\section{Conclusion}

\begin{frame}{Conclusion}
  \begin{itemize}
    \item A distributed network of stand-alone acoustic sensors with a remote server
          detects, classifies and localizes impulsive sounds in 2D.
    \item Cepstral coefficients are an effective feature extractor for gunshot
          classification with both NN and SVM.
    \item GTCC beats Mel-frequency based coefficients; NN achieves slightly better
          classification success, especially with longer frames.
    \item Validated with various firearm calibers and ammunition.
  \end{itemize}

\begin{frame}{Appendix -- other cepstral coefficients}
  \begin{itemize}
    \item \textbf{LPCC} (Linear Prediction Cepstral Coefficients) and \textbf{PLPC}
          (Perceptual Linear Prediction Coefficients) -- noise elimination, speech
          recognition.
    \item \textbf{GFCC} (Greenwood Function Cepstral Coefficients) -- derived from
          human auditory response models, used for classifying animal sounds.
    \item \textbf{GTCC} (Gammatone-Frequency Cepstral Coefficients) -- same family,
          greater resistance to noise than MFCC.
  \end{itemize}
\end{frame}

\begin{frame}{Appendix -- metrics}
  \begin{description}
    \item[ACC] Accuracy -- share of correctly classified events.
    \item[RLC] Recall, also known as \emph{sensitivity} -- share of actual gunshots
      of the class that were detected.
    \item[MCC] Matthews Correlation Coefficient -- balanced measure robust to class
      imbalance; $+1$ perfect, $0$ random.
  \end{description}
  \vfill
  {\small\color{gray} Definitions as used in the source study: Svatoš \& Holub,
   \emph{Meas. Sci. Technol.} \textbf{35} (2024) 076122, eqs.~(1)--(3).}
\end{frame}

{\small 10 min --- target \textbf{9:30} --- 2 clicks}

\cue{0:00}{Title}{%
  \say{Good morning. My name is Martin Maxa, from the Department of Measurement at the
  Czech Technical University in Prague. This is joint work with Doctor Jakub Svatoš.}
  \bt subject: detection, classification, localization of impulsive sounds ---
  gunshots as the target class
  \bt \textbf{core of the contribution:} comparison of cepstral extractors $+$ two
  classifiers; localization $=$ second function of the same network}

\cue{0:45}{Motivation}{%
  \say{The practical motivation --- the incidence of shootings in public spaces --- is
  stated in the abstract; I will concentrate on the measurement problem.}
  \bt acoustic channel complements cameras: \textbf{omnidirectional}, \textbf{no line of sight}
  \bt \textbf{[CUT]} three functions: detection, classification (to calibre), localization ---
  in noisy, reverberant outdoor environments
  \bt \textbf{three questions:} (1) which cepstral representation separates gunshots from
  similar false alarms, (2) NN or SVM on those features, (3) frame length vs.\ reflections}

\cue{1:50}{Signature}{%
  \say{The acoustic signature of a gunshot has two components, shown here on a real
  recording.}
  \bt \textbf{shock wave first}, $\approx$2.5\,ms --- supersonic projectiles only ---
  \textbf{[CUT]} Mach cone along the trajectory, N-shaped pressure profile
  \bt \textbf{\color{accent}[CLICK]} muzzle blast, \emph{in this recording} $\approx$45\,ms
  later --- propellant gases at the barrel, speed of sound, 1--3\,ms
  \bt between and after $=$ \textbf{reflections}, not source signature ---
  \say{relevant to frame length; I return to it in the results}}

\cue{3:00}{Single-domain}{%
  \say{Neither signal domain is sufficient on its own.}
  \bt spectral analysis of a frame $\to$ characterises the environment, not the source
  \bt time-domain pattern $\to$ nonlinear dispersion, obstacles in the path
  \bt $\Rightarrow$ combine both domains --- which is what a cepstral representation does}

\cue{3:30}{System}{%
  \say{The system architecture is a distributed network of stand-alone sensor units and
  a remote server.}
  \bt unit monitors continuously $\to$ on an impulsive event, transmits the segment
  \bt server: \textbf{classification} $+$ \textbf{2D position} from several units
  \bt several units: coverage --- and a 2D position needs \emph{at least two bearings}}

\cue{4:20}{Localize}{%
  \say{Localization. Each sensor unit carries two MEMS microphones at a spacing of
  0.186 metres, sampled at 44.1 kilohertz.}
  \bt TDoA between channels, at most $\approx$24 samples for this geometry
  \bt $\Theta = \arcsin\!\big(c\,\Delta n / (d f_s)\big)$ --- speed of sound $\times$ delay,
  over spacing $\times$ sampling rate
  \bt \textbf{\color{accent}[CLICK]} one unit $=$ a direction; two bearings intersect $=$
  position estimate
  \bt evaluated on \textbf{70 events}, 9 angles, $-45^\circ$ to $+60^\circ$ ---
  cross-corr $+$ parabolic sub-sample interpolation, \textbf{MAE $0.77^\circ$}}

\cue{5:20}{Features}{%
  \say{Classification. All four feature extractors share the cepstral processing chain:
  framing, FFT, power spectrum, logarithm, discrete cosine transform.}
  \bt \textbf{they differ in the filter bank} --- power spectrum $\to$ bank $\to$ log ---
  not in the diagram
  \bt MFCC mel (speech; validated on ranges) $|$ LFCC linear $|$ IMFCC inverted mel $|$
  \textbf{\color{accent}GTCC gammatone}, auditory model, reported more noise-robust
  \bt grid: 4 extractors $\times$ 3 frame lengths (15/30/50\,ms) $\times$ 2 classifiers}

\cue{6:25}{\color{accent}TABLE}{%
  \say{Table one gives the results for the nine-millimetre class.}
  \bt rows: frame length, four extractors within $|$ columns: ACC, RLC, MCC $\times$ SVM, NN
  \bt \textbf{GTCC rows highlighted --- highest value of all three metrics in every group,
  both classifiers}
  \bt best: \textbf{GTCC $+$ NN, 50\,ms $\to$ 99.15 \,/\, 96.77 \,/\, 0.98}
  \bt \textbf{for scale:} 118 test events, 31 of them 9\,mm $\Rightarrow$ \textbf{117 of 118}
  correct, \textbf{30 of 31} recalled}

\cue{7:35}{Discussion}{%
  \say{Three observations.}
  \bt MFCC, LFCC, IMFCC comparable --- mel / linear / inverted-mel spacing has little effect
  \bt \textbf{GTCC consistently better}, every frame length, both classifiers ---
  \textbf{[CUT]} auditory model, known noise-robust: \emph{probable} explanation, mechanism
  not isolated here
  \bt \textbf{NN slightly ahead of SVM}, particularly at longer frames --- \emph{tendency
  across the study, not every row} (LFCC 30/50\,ms goes the other way)
  \bt longer frames help, \emph{but} 50\,ms increasingly contains the impulse response of
  the surroundings --- \say{the trend cannot be extrapolated further}}

\cue{8:45}{Conclusion}{%
  \say{To conclude.}
  \bt distributed network $+$ server $\to$ detection, classification, 2D localization
  \bt cepstral coefficients effective with both classifiers tested
  \bt GTCC $>$ mel family; NN slightly higher, particularly at longer frames
  \bt results on real firearms, several calibres and ammunition types}

\cue{9:30}{}{\say{Thank you.}}
